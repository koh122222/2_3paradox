TRY_REPLACE:hire_privateer = {
	type = owncountry

	potential = {
		trigger_if = {
			limit = { scope:actor = { is_ai = yes } }
			scope:actor = { sailors > 1000 }
		}
	}

	allow = {
		scope:actor = {
			modifier:can_hire_privateers = yes
			# The price definition handles the first 100 gold. The action
			# also checks the dynamically calculated surcharge before hiring.
			gold >= mrb13_privateer_hire_gold
		}
		scope:target ?= {
			not = { has_privateers_from = scope:actor }
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 2
	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = privateers
	price = price:hire_privateer

	select_trigger = {
		looking_for_a = area
		target_flag = target
		name = "hire_privateer_select_area"
		none_available_msg_key = "hire_privateer_no_areas"
		column = { data = name }
		visible = {
			within_naval_range_of = scope:actor
			is_area_coastal_sea = yes
			any_adjacent_ports_to_area = {
				or = {
					has_owner = no
					and = {
						owner = { save_temporary_scope_as = location_owner }
						not = {
							scope:actor = {
								giving_scripted_relation = {
									type = relation_type:anti_piracy_agreement
									target = scope:location_owner
								}
							}
						}
					}
				}
			}
		}
	}

	effect = {
		scope:actor = {
			# add_gold does not evaluate a named calculated script value here,
			# so the surcharge is kept inline.
			add_gold = {
				value = {
					value = 0
					if = {
						limit = { current_age_or_later = { age = age_6_revolutions } }
						add = { value = current_year subtract = 1337 multiply = 5 }
					}
					else_if = {
						limit = { current_age_or_later = { age = age_5_absolutism } }
						add = { value = current_year subtract = 1337 multiply = 4 }
					}
					else_if = {
						limit = { current_age_or_later = { age = age_4_reformation } }
						add = { value = current_year subtract = 1337 multiply = 3 }
					}
					else_if = {
						limit = { current_age_or_later = { age = age_3_discovery } }
						add = { value = current_year subtract = 1337 multiply = 2 }
					}
					else_if = {
						limit = { current_age_or_later = { age = age_2_renaissance } }
						add = { value = current_year subtract = 1337 }
					}
				}
				multiply = -1
			}
		}
		if = {
			limit = { exists = scope:target }
			scope:target = { hire_privateer = scope:actor }
		}
	}

	ai_will_do = {
		add = {
			if = {
				limit = { exists = scope:target }
				value = "scope:target.privateer_utility(scope:actor)"
			}
		}
	}
}
