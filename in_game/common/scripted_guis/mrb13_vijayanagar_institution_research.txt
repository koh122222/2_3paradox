mrb13_asia_research_toggle_population = {
	scope = country
	is_shown = { has_or_had_tag = VIJ }
	is_valid = { NOT = { has_variable = mrb13_asia_research_population_sacrifice_cooldown } }
	effect = {
		root = {
			if = {
				limit = { NOT = { has_variable = mrb13_asia_research_population_sacrifice_cooldown } }
				if = { limit = { has_variable = mrb13_asia_research_population_sacrifice } remove_variable = mrb13_asia_research_population_sacrifice }
				else = { set_variable = { name = mrb13_asia_research_population_sacrifice value = 1 } }
				set_variable = { name = mrb13_asia_research_population_sacrifice_cooldown value = 1 months = 1 }
			}
		}
	}
}
