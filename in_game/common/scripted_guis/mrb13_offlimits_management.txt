mrb13_offlimits_toggle_france = {
	scope = country
	effect = {
		root = {
			if = {
				limit = { has_variable = mrb13_offlimits_france_active }
				every_country = {
					limit = { has_or_had_tag = FRA }
					remove_country_modifier = mrb13_france_offlimits
				}
				remove_variable = mrb13_offlimits_france_active
			}
			else = {
				every_country = {
					limit = { has_or_had_tag = FRA }
					add_country_modifier = { modifier = mrb13_france_offlimits months = -1 mode = replace }
				}
				set_variable = { name = mrb13_offlimits_france_active value = 1 }
			}
		}
	}
}
