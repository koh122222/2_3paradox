mrb2_ormuz_annex_country_with_cores_effect = {
	if = {
		limit = { country_exists = c:$tag$ }
		c:$tag$ = {
			every_owned_location = {
				add_core = c:ORM
			}
		}
		annex_country = {
			country = c:$tag$
			reason = MilitaryConquest
			transfer = yes
		}
	}
}

mrb2_ormuz_release_jarwanids_effect = {
	if = {
		limit = {
			country_exists = c:JRW
			c:JRW = { is_subject_of = root }
		}
		cancel_subject = c:JRW
	}
	if = {
		limit = { country_exists = c:JRW }
		every_owned_location = {
			limit = {
				region = region:arabia_region
			}
			add_core = c:JRW
			change_location_owner = c:JRW
		}
	}
}

mrb2_annex_persian_region_effect = {
	every_country = {
		limit = {
			NOT = { this = root }
			any_owned_location = {
				region = region:persia_region
			}
		}
		every_owned_location = {
			limit = {
				region = region:persia_region
			}
			add_core = root
			change_location_owner = root
			change_integration_level = core
		}
	}
}

# Direct subjects count at half weight. This is called on the second stage and
# then once per year, so it never participates in a monthly country-wide scan.
mrb2_update_persian_administration_control_effect = {
	set_variable = { name = mrb2_persia_weighted_population value = 0 }
	set_variable = { name = mrb2_persia_iranian_population value = 0 }
	set_variable = { name = mrb2_persia_owned_population value = 0 }
	set_variable = { name = mrb2_persia_owned_iranian_population value = 0 }
	set_variable = { name = mrb2_persia_subject_population value = 0 }
	set_variable = { name = mrb2_persia_subject_iranian_population value = 0 }

	every_owned_location = {
			every_pop = {
			root = {
				change_variable = {
					name = mrb2_persia_weighted_population
					add = prev.pop_size
				}
				change_variable = {
					name = mrb2_persia_owned_population
					add = prev.pop_size
				}
			}
			if = {
				limit = { culture = { has_culture_group = culture_group:iranian_group } }
				root = {
					change_variable = {
						name = mrb2_persia_iranian_population
						add = prev.pop_size
					}
					change_variable = {
						name = mrb2_persia_owned_iranian_population
						add = prev.pop_size
					}
				}
			}
		}
	}

	every_subject = {
		every_owned_location = {
			every_pop = {
				root = {
					change_variable = {
						name = mrb2_persia_weighted_population
						add = { value = prev.pop_size multiply = 0.5 }
					}
					change_variable = {
						name = mrb2_persia_subject_population
						add = prev.pop_size
					}
				}
				if = {
					limit = { culture = { has_culture_group = culture_group:iranian_group } }
					root = {
						change_variable = {
							name = mrb2_persia_iranian_population
							add = { value = prev.pop_size multiply = 0.5 }
						}
						change_variable = {
							name = mrb2_persia_subject_iranian_population
							add = prev.pop_size
						}
					}
				}
			}
		}
	}

	# Store the share as 0..1, the standard percentage format used by the UI.
	set_variable = { name = mrb2_persia_iranian_population_percent value = 0 }
	if = {
		limit = { var:mrb2_persia_weighted_population > 0 }
		set_variable = {
			name = mrb2_persia_iranian_population_percent
			value = {
				value = var:mrb2_persia_iranian_population
				divide = var:mrb2_persia_weighted_population
				min = 0
				max = 1
			}
		}
	}

	# 60% gives no bonus; 80% or more restores the full +20% maximum control.
	set_variable = {
		name = mrb2_persia_restoration_control_bonus
		value = {
			value = var:mrb2_persia_iranian_population_percent
			subtract = 0.60
			divide = 0.20
			multiply = 0.20
			min = 0
			max = 0.20
		}
	}

	# The law's location modifier is evaluated in location scope. Cache the country
	# result on each Persian location, then let the law apply it dynamically.
	every_owned_location = {
		limit = { region = region:persia_region }
		set_variable = {
			name = mrb2_persia_restoration_control_location_scale
			value = root.var:mrb2_persia_restoration_control_bonus
		}
		remove_location_modifier = mrb2_persian_administration_control_modifier
	}

	set_variable = { name = mrb2_persia_administration_calculation_version value = 9 }
}
