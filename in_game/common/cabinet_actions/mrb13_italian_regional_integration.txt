mrb13_italian_regional_integration = {
	potential = {
		culture = { has_culture_group = culture_group:italian_group }
		total_population < 5000
		any_province = {
			any_location_in_province = {
				integration_level = conquered
				region = region:italy_region
			}
		}
	}

	ability = adm
	icon = integrate_area
	allow_multiple = yes

	allow = {
		total_population < 5000
		any_province = {
			any_location_in_province = {
				integration_level = conquered
				region = region:italy_region
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		name = "integrate_province_select_province"
		none_available_msg_key = "integrate_province_no_provinces"
		column = { data = name }
		column = { data = integration }
		column = { data = population }
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = mrb13_italian_regional_integration interaction_target = target } }
			any_location_in_province = {
				integration_level = conquered
				region = region:italy_region
			}
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = integration_progress
			}
		}
	}

	is_finished = {
		OR = {
			scope:actor = { total_population >= 5000 }
			scope:target ?= {
				any_location_in_province = {
					OR = {
						integration_level = integrated
						integration_level = core
					}
					count = all
				}
			}
		}
	}

	province_modifier = {
		province_integration_speed = 2.5
	}

	progress = {
		scope:target ?= {
			value = province_average_integration
		}
	}

	map_marker = {
		show_for_owner = yes
	}
}
