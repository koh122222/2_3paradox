mrb13_moscow_regional_integration = {
	potential = {
		has_or_had_tag = MOS
		any_province = {
			any_location_in_province = {
				integration_level = conquered
				OR = {
					region = region:russian_region
					region = region:steppes_region
					region = region:ural_region
					region = region:west_siberia_region
					region = region:east_siberia_region
				}
			}
		}
	}

	ability = adm
	icon = integrate_area
	allow_multiple = yes

	allow = {
		any_province = {
			any_location_in_province = {
				integration_level = conquered
				OR = {
					region = region:russian_region
					region = region:steppes_region
					region = region:ural_region
					region = region:west_siberia_region
					region = region:east_siberia_region
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		name = "integrate_province_select_province"
		none_available_msg_key = "integrate_province_no_provinces"
		column = { data = name }
		column = { data = integration }
		column = { data = population }
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = mrb13_moscow_regional_integration interaction_target = target } }
			any_location_in_province = {
				integration_level = conquered
				OR = {
					region = region:russian_region
					region = region:steppes_region
					region = region:ural_region
					region = region:west_siberia_region
					region = region:east_siberia_region
				}
			}
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = integration_progress
			}
		}
	}

	is_finished = {
		scope:target ?= {
			any_location_in_province = {
				OR = {
					integration_level = integrated
					integration_level = core
				}
				count = all
			}
		}
	}

	province_modifier = {
		province_integration_speed = 2
		local_pop_assimilation_speed = 0.03
		local_pop_conversion_speed = 0.03
		local_max_control = 0.3
		local_monthly_control = 0.025
	}

	progress = {
		scope:target ?= {
			value = province_average_integration
		}
	}

	map_marker = {
		show_for_owner = yes
	}
}
