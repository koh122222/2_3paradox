mrb2_persia.2 = {
	type = country_event
	fire_only_once = yes
	title = mrb2_persia.2.title
	desc = mrb2_persia.2.desc
	outcome = positive

	illustration_tags = {
		10 = regular
		10 = interior
	}

	trigger = {
		tag = IRA
		has_or_had_tag = ORM
	}

	immediate = {
		create_character = {
			age = { 30 50 }
			adm = { 50 80 }
			dip = { 50 80 }
			mil = { 50 80 }
			estate = estate_type:clergy_estate
			culture = root.culture
			religion = root.religion
		}
	}

	option = {
		name = mrb2_persia.2.a
		change_religion = religion:shia
		change_religion_for_ruler_and_family = { country = root religion = religion:shia }
		add_country_modifier = {
			modifier = mrb2_persia_shia_transition
			years = 20
			mode = replace
		}
	}

	option = {
		name = mrb2_persia.2.b
	}
}

mrb2_persia.4 = {
	type = country_event
	hidden = yes
	title = empty_text
	desc = empty_text

	trigger = {
		tag = IRA
		has_or_had_tag = ORM
		has_policy = mrb2_persian_restoration_policy
		OR = {
			NOT = { has_variable = mrb2_persia_administration_calculation_version }
			var:mrb2_persia_administration_calculation_version < 9
		}
	}

	immediate = {
		mrb2_update_persian_administration_control_effect = yes
		lock_policy_effect = { type = mrb2_persian_unification_policy }
		lock_policy_effect = { type = mrb2_persian_restoration_policy }
	}

	option = { name = empty_text }
}
